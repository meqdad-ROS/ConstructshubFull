\newcounter{example}[chapter]
\renewcommand{\theexample}{\thechapter.\arabic{example}}
% The counter is stepped OUTSIDE the box, so a \label placed inside the
% example body refers to the example number (not the current section).
\newtcolorbox{examplebox}[1]{enhanced,breakable,colback=white,colframe=NavyBlue!70!black,
  fonttitle=\bfseries,title={\examplename~\theexample\ifx&#1&\else: #1\fi}}
\newenvironment{example}[1][]{\refstepcounter{example}\begin{examplebox}{#1}}{\end{examplebox}}
\AtBeginDocument{\ifdefined\crefname\crefname{example}{Example}{Examples}\fi}

\newtcolorbox{practicenote}{enhanced,breakable,colback=Goldenrod!8,colframe=Goldenrod!80!black,
  fonttitle=\bfseries,title=\practicenotename}
